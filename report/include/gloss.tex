%--- Single rigid body on a Lie group (minimal coordinates) ---------------
\newglossaryentry{sym:g}{type=symbols,
	sort={04},
	name={\ensuremath{g_k\in G}},
	description={configuration of a \emph{single} rigid body, represented directly as an element of the matrix Lie group \(G\) (e.g.\ \(SO(3)\)); this is the \textbf{minimal-coordinate} representation used for the unconstrained 3D pendulum. Contrast with \(\bm{q}\in\bm{Q}_{\mathrm{max}}\), the maximal-coordinate configuration of a constrained multi-body system}
}

\newglossaryentry{sym:f}{type=symbols,
	sort={05},
	name={\ensuremath{f_k\in G}},
	description={relative (incremental) Lie group element between two time steps, \(g_{k+1}=g_kf_k\)}
}

\newglossaryentry{sym:Ltrans}{type=symbols,
	sort={05a},
	name={\ensuremath{\mathbb{L}_h}},
	description={left-translation map, \(\mathbb{L}_h(g)=hg\); its differential \(T_e\mathbb{L}_g\) and dual \(T_e^*\mathbb{L}_g\) appear throughout the derivative computations. Blackboard-bold distinguishes it from the Lagrangian \(L\), \(L_d\)}
}

\newglossaryentry{sym:eta}{type=symbols,
	sort={06},
	name={\ensuremath{\eta_k,X_k}},
	description={Lie-algebra-valued infinitesimal variations of \(g_k\) and \(f_k\), \(\delta g_k=g_k\eta_k\)}
}

\newglossaryentry{sym:mu}{type=symbols,
	sort={07},
	name={\ensuremath{\mu_k\in\mathfrak{g}^*}},
	description={discrete (body-fixed) momentum, obtained from the discrete Legendre transform \(\mathbb{F}^{\pm}L_d\) of the discrete Lagrangian; used generically for arbitrary \(G\), forced or unforced. Contrast with \(\bm{\Pi}_k\), its concrete specialization once \(G=SO(d)\)}
}

%--- Constrained multi-body system (maximal coordinates) ------------------
\newglossaryentry{sym:q}{type=symbols,
	sort={08},
	name={\ensuremath{\bm{q}\in\bm{Q}_{\mathrm{max}}}},
	description={stacked configuration of a \emph{constrained multi-body} system in \textbf{maximal coordinates}, \(\bm{q}=(\bm{x}_1,\bm{R}_1,\dots,\bm{x}_{n_b},\bm{R}_{n_b})\); used for the Acrobot. Contrast with \(g\in G\), the minimal-coordinate configuration of a single unconstrained body}
}

\newglossaryentry{sym:Qmax}{type=symbols,
	sort={09},
	name={\ensuremath{\bm{Q}_{\mathrm{max}}}},
	description={maximal-coordinate configuration manifold of a multi-body system, \(\bm{Q}_{\mathrm{max}}=\prod_{i=1}^{n_b}\left(\mathbb{R}^d\times SO(d)\right)\)}
}

\newglossaryentry{sym:nb}{type=symbols,
	sort={10},
	name={\ensuremath{n_b}},
	description={number of rigid bodies in a multi-body system}
}

\newglossaryentry{sym:d}{type=symbols,
	sort={11},
	name={\ensuremath{d}},
	description={spatial dimension of the ambient space, \(d\in\{2,3\}\)}
}

\newglossaryentry{sym:xi}{type=symbols,
	sort={12},
	name={\ensuremath{\bm{x}_i,\bm{R}_i}},
	description={center-of-mass position and orientation of rigid body \(i\), the per-body components of \(\bm{q}\)}
}

\newglossaryentry{sym:phi}{type=symbols,
	sort={13},
	name={\ensuremath{\bm{\phi}(\bm{q})}},
	description={holonomic joint-constraint map, \(\bm{\phi}:\bm{Q}_{\mathrm{max}}\rightarrow\mathbb{R}^{n_c}\), enforced via \(\bm{\phi}(\bm{q})=\bm{0}\)}

\newglossaryentry{sym:nc}{type=symbols,

\newglossaryentry{sym:lambda}{type=symbols,

\newglossaryentry{sym:rho}{type=symbols,

\newglossaryentry{sym:p0}{type=symbols,

	sort={23c},
	name={\ensuremath{\Theta_L,\Omega_L}},

\newglossaryentry{sym:U}{type=symbols,
	sort={23a},
	name={\ensuremath{U(g)}},
	description={potential energy of the \emph{generic}, abstract single rigid body on a Lie group \(G\), as used in the general Lagrangian/Legendre-transform theory before specializing to a concrete system}
}

\newglossaryentry{sym:V}{type=symbols,
	sort={23b},
	name={\ensuremath{V,V_R,V_T}},
	description={potential energy of a \emph{concrete} mechanical system (3D pendulum, Acrobot); \(V_R\), \(V_T\) split the Acrobot's potential into rotational and translational contributions. Contrast with the generic \(U(g)\)}

\newglossaryentry{sym:J}{type=symbols,
	sort={24},
	name={\ensuremath{\bm{J}}},

\newglossaryentry{sym:Jd}{type=symbols,
	sort={25},

}

\newglossaryentry{sym:Mi}{type=symbols,
	sort={26},

}

\newglossaryentry{sym:Pi}{type=symbols,
	sort={27},
	name={\ensuremath{\bm{\Pi}_k}},
	description={discrete body-fixed angular momentum, \(\bm{\Pi}_k=\bm{J}\bm{\Omega}_k\); the concrete \(SO(d)\)-specialization of the generic momentum \(\mu_k\) (same discrete Legendre transform, applied to the 3D pendulum or an Acrobot link), analogous to \(g_k\) specializing to \(\bm{R}_k\)}
}

%--- Control ---------------------------------------------------------------
\newglossaryentry{sym:u}{type=symbols,
	sort={28},
	name={\ensuremath{\bm{u},\bm{u}_k}},
	description={continuous / discrete control input (generalized force), acting through \(\mathfrak{g}^*\); used as-is whenever a system has only one type of degree of freedom (e.g.\ the 3D pendulum). For a multi-body system, splits per body into a translational part \(\bm{u}_i^x\) and a rotational part \(\bm{\tau}_i\)}
}

\newglossaryentry{sym:ux}{type=symbols,
	sort={28a},
	name={\ensuremath{\bm{u}_i^x}},
	description={translational component of the generalized force on body \(i\), dual to \(\bm{x}_i\); zero for the Acrobot, which is actuated only at the joint}
}

\newglossaryentry{sym:tau}{type=symbols,
	sort={28b},
	name={\ensuremath{\bm{\tau}_k,\bm{\tau}_g}},
	description={rotational component of the generalized force (torque), dual to \(\bm{R}_i\); \(\bm{\tau}_k\) is the Acrobot's generalized joint (control) torque. \(\bm{\tau}_g(\bm{g}_k)\) is the potential-energy-derived generalized torque -- generic (arbitrary \(G\)) in the forced discrete Euler-Lagrange equations, and \(\bm{\tau}_{g,k}=mg\,\bm{\rho}_c\times\bm{R}_k^\top\bm{e}_3\) for the concrete 3D pendulum -- matching the gravity torque \(\tau_{g,k}\) of the reduced Newton-Euler model in the Appendix. Replaces the earlier \(\bm{M}(\bm{g}_k)\)/\(\bm{M}_k\) notation, which clashed with the mass matrix \(\bm{M}_i\)}
}

\newglossaryentry{sym:umax}{type=symbols,
	sort={29},
	name={\ensuremath{u_{\max}}},
	description={maximum admissible magnitude of the control input}
}

\newglossaryentry{sym:x}{type=symbols,
	sort={30},
	name={\ensuremath{\bm{x},\bm{x}_k}},
	description={state vector used in the optimal control / MPC formulation, stacking configuration and velocity or momentum variables}
}
